from pathlib import Path
import json, math
root=Path('<WORKSPACE>')
p=json.loads((root/'.local/chain-sewing-acceptance-20261114/fixtures/asymmetric-fine.weftforge').read_text())['project']
points={v['id']:v['position'] for v in p['points']}
curves={v['id']:v for v in p['curves']}
def coeff(c):
    a,b=points[c['start']],points[c['end']]
    if c['controls'] is None:return [[a[k],b[k]-a[k]] for k in range(2)]
    u,v=c['controls']
    return [[a[k],3*(u[k]-a[k]),3*(a[k]-2*u[k]+v[k]),-a[k]+3*u[k]-3*v[k]+b[k]] for k in range(2)]
def curve_area(c):
    x,y=coeff(c)
    return 0.5*sum((x[i]*j*y[j]-y[i]*j*x[j])/(i+j) for i in range(len(x)) for j in range(1,len(x)))
assert abs(sum(curve_area(c) for c in [{'start':3,'end':4,'controls':None},{'start':4,'end':5,'controls':None},{'start':5,'end':6,'controls':None},{'start':6,'end':3,'controls':None}])-1.47175)<1e-12
L=1.4*p['resolution']
maxarea=math.sqrt(3)/4*L*L
pieces=[{'piece':v['id'],'analytic_area_m2':abs(sum(curve_area(curves[c]) for c in v['boundary']))} for v in p['pieces']]
total=sum(v['analytic_area_m2'] for v in pieces)
for v in pieces:v['area_edge_triangle_lower_bound']=math.ceil(v['analytic_area_m2']/maxarea)
result={'resolution_m':p['resolution'],'max_edge_m':L,'max_triangle_area_m2':maxarea,'pieces':pieces,'total_analytic_area_m2':total,'total_area_edge_triangle_lower_bound':math.ceil(total/maxarea),'mesh_vertex_limit':100000,'mesh_triangle_limit':200000,'interpretation':'Area bound is necessary, not a sufficient construction or proof of feasibility.'}
(root/'.local/chain-fine-mesh-20261114/area-bounds.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result,indent=2))
